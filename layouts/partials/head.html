{{/* Fingerprinted: the filename carries a hash of the contents, so editing custom.css publishes it at a new URL. Cloudflare Pages caches static assets in the browser for 4 hours (max-age=14400) while HTML is always revalidated, so without this a returning visitor gets new HTML with a stale stylesheet until that expires. */}} {{ $custom := resources.Get "css/custom.css" | fingerprint "sha256" -}} {{ $title := print .Site.Title " | " .Title }} {{ if .IsHome }}{{ $title = .Site.Title }}{{ end }} {{ $title }} {{/* Feed autodiscovery: lets a reader find the feed when someone pastes a page URL instead of the feed URL. $siteFeed is the whole-site feed; $pageFeed is this page's own feed (section and tag pages have one), emitted only when it is a different URL. */}} {{ $siteFeed := .Site.Home.OutputFormats.Get "rss" }} {{ $pageFeed := .OutputFormats.Get "rss" }} {{ with $pageFeed }} {{ if or (not $siteFeed) (ne .Permalink $siteFeed.Permalink) }} {{ end }} {{ end }} {{ with $siteFeed }} {{ end }}
